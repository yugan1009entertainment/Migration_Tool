// with 40C_75RH special branch, cutoff split, WEEKDAY(x, 2) Mon=1 mode

Power BI DAX:
─────────────
measure 'Test Completion Time m' =

VAR ActualDate   = SELECTEDVALUE('bdy_qc_stab_main'[Actual Pull Date_])
VAR ReviewedDate = SELECTEDVALUE('bdy_qc_stab_main'[Date Reviewed Test])
VAR ConditionVal = SELECTEDVALUE('bdy_qc_stab_main'[Condition])

VAR TodayDate    = TODAY()
VAR EndDate      = IF(ISBLANK(ReviewedDate), TodayDate, ReviewedDate)

RETURN

    ISBLANK(ActualDate),
    BLANK(),
    IF(
        ActualDate <= CutOffDate,
        SWITCH(
            TRUE(),
            ConditionVal = "40C_75RH" && NOT ISBLANK(EndDate),
            VAR StartWD  = WEEKDAY(ActualDate, 2)
            VAR EndWD    = WEEKDAY(EndDate, 2)
            VAR TotalDays= DATEDIFF(ActualDate, EndDate, DAY)
            RETURN
                DIVIDE(TotalDays - (7 - StartWD) - EndWD, 7) * 5
                + MIN(5, 7 - StartWD)
                + MIN(5, EndWD - 1)
                - 1,

            NOT ISBLANK(EndDate),
            VAR TotalDays2    = DATEDIFF(ActualDate, EndDate, DAY)
            VAR Weeks         = INT(TotalDays2 / 7)
            VAR ExtraDays     = MOD(TotalDays2, 7)
            VAR StartWD2      = WEEKDAY(ActualDate, 2)
            VAR WeekendAdjust = MAX(0, StartWD2 + ExtraDays - 5)
            RETURN
                (Weeks * 5) + MIN(5, ExtraDays) - WeekendAdjust
        ),
        IF(
            ISBLANK(ReviewedDate),
            DATEDIFF(ActualDate, TodayDate, DAY),
            DATEDIFF(ActualDate, ReviewedDate, DAY)
        )
    )

### PATTERN-03 | Working-day calculation (manual weekday arithmetic)
─────────────────────────────────────────────────────────────
Tableau uses DATEDIFF("weekday", ...) or custom weekday formula.
Power BI uses manual weekday arithmetic with WEEKDAY(x, 2) (Mon=1):
  VAR StartWD = WEEKDAY(StartDate, 2)
  VAR EndWD   = WEEKDAY(EndDate, 2)
  VAR TotalDays = DATEDIFF(StartDate, EndDate, DAY)
  RETURN DIVIDE(TotalDays - (7 - StartWD) - EndWD, 7) * 5
         + MIN(5, 7 - StartWD) + MIN(5, EndWD - 1) - 1
For simple NETWORKDAYS use: NETWORKDAYS(Start, End, 1) - 1
</ch>
